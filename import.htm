import 'package:flutter/material.dart';

 

import 'game_logic.dart';

 

void main() => runApp(const CaroApp());

 

class CaroApp extends StatelessWidget {

  const CaroApp({super.key});

 

  @override

  Widget build(BuildContext context) {

    return MaterialApp(

      title: 'Caro',

      debugShowCheckedModeBanner: false,

      theme: ThemeData(

        colorSchemeSeed: Colors.indigo,

        useMaterial3: true,

      ),

      home: const GameScreen(),

    );

  }

}

 

class GameScreen extends StatefulWidget {

  const GameScreen({super.key});

 

  @override

  State<GameScreen> createState() => _GameScreenState();

}

 

class _GameScreenState extends State<GameScreen> {

  final CaroGame game = CaroGame();

  bool vsComputer = true;

  bool aiThinking = false;

  int scoreX = 0;

  int scoreO = 0;

 

  // Khi chơi với máy: người chơi là X, máy là O.

  static const Player human = Player.x;

  static const Player ai = Player.o;

 

  void _onTap(int r, int c) {

    if (aiThinking) return;

    if (vsComputer && game.current != human) return;

    setState(() {

      if (!game.play(r, c)) return;

      _afterMove();

    });

    if (vsComputer && !game.isOver && game.current == ai) _aiMove();

  }

 

  Future<void> _aiMove() async {

    setState(() => aiThinking = true);

    await Future.delayed(const Duration(milliseconds: 350));

    if (!mounted) return;

    final move = game.bestMove(ai);

    setState(() {

      game.play(move.x, move.y);

      aiThinking = false;

      _afterMove();

    });

  }

 

  void _afterMove() {

    if (game.winner == Player.x) scoreX++;

    if (game.winner == Player.o) scoreO++;

    if (game.isOver) {

      WidgetsBinding.instance.addPostFrameCallback((_) => _showResult());

    }

  }

 

  void _showResult() {

    final String message;

    if (game.isDraw) {

      message = 'Hòa!';

    } else if (vsComputer) {

      message = game.winner == human ? 'Bạn thắng! 🎉' : 'Máy thắng!';

    } else {

      message = '${game.winner.symbol} thắng! 🎉';

    }

    showDialog(

      context: context,

      builder: (ctx) => AlertDialog(

        title: const Text('Kết thúc'),

        content: Text(message, style: const TextStyle(fontSize: 20)),

        actions: [

          TextButton(

            onPressed: () => Navigator.pop(ctx),

            child: const Text('Xem bàn cờ'),

          ),

          FilledButton(

            onPressed: () {

              Navigator.pop(ctx);

              _newGame();

            },

            child: const Text('Ván mới'),

          ),

        ],

      ),

    );

  }

 

  void _newGame() {

    if (aiThinking) return;

    setState(game.reset);

  }

 

  void _undo() {

    if (aiThinking || game.history.isEmpty) return;

    setState(() {

      if (game.winner == Player.x) scoreX--;

      if (game.winner == Player.o) scoreO--;

      game.undo();

      // Khi chơi với máy, lùi luôn nước của máy để quay về lượt người chơi.

      if (vsComputer && game.current == ai && game.history.isNotEmpty) {

        game.undo();

      }

    });

  }

 

  void _toggleMode(bool value) {

    if (aiThinking) return;

    setState(() {

      vsComputer = value;

      scoreX = 0;

      scoreO = 0;

      game.reset();

    });

  }

 

  String get _status {

    if (game.isDraw) return 'Hòa!';

    if (game.winner != Player.none) return '${game.winner.symbol} thắng!';

    if (aiThinking) return 'Máy đang nghĩ...';

    if (vsComputer) {

      return game.current == human ? 'Lượt của bạn (X)' : 'Lượt của máy (O)';

    }

    return 'Lượt của ${game.current.symbol}';

  }

 

  @override

  Widget build(BuildContext context) {

    final scheme = Theme.of(context).colorScheme;

    return Scaffold(

      appBar: AppBar(

        title: const Text('Caro'),

        centerTitle: true,

        actions: [

          IconButton(

            tooltip: 'Đi lại',

            icon: const Icon(Icons.undo),

            onPressed: _undo,

          ),

          IconButton(

            tooltip: 'Ván mới',

            icon: const Icon(Icons.refresh),

            onPressed: _newGame,

          ),

        ],

      ),

      body: SafeArea(

        child: Column(

          children: [

            Padding(

              padding: const EdgeInsets.fromLTRB(16, 8, 16, 0),

              child: SegmentedButton<bool>(

                segments: const [

                  ButtonSegment(

                    value: true,

                    icon: Icon(Icons.smart_toy),

                    label: Text('Chơi với máy'),

                  ),

                  ButtonSegment(

                    value: false,

                    icon: Icon(Icons.people),

                    label: Text('2 người'),

                  ),

                ],

                selected: {vsComputer},

                onSelectionChanged: (s) => _toggleMode(s.first),

              ),

            ),

            Padding(

              padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 12),

              child: Row(

                mainAxisAlignment: MainAxisAlignment.spaceBetween,

                children: [

                  _ScoreChip(

                    label: vsComputer ? 'Bạn (X)' : 'X',

                    score: scoreX,

                    color: Colors.blue,

                    active: !game.isOver && game.current == Player.x,

                  ),

                  Flexible(

                    child: Text(

                      _status,

                      textAlign: TextAlign.center,

                      style: Theme.of(context).textTheme.titleMedium,

                    ),

                  ),

                  _ScoreChip(

                    label: vsComputer ? 'Máy (O)' : 'O',

                    score: scoreO,

                    color: Colors.red,

                    active: !game.isOver && game.current == Player.o,

                  ),

                ],

              ),

            ),

            Expanded(

              child: Center(

                child: Padding(

                  padding: const EdgeInsets.all(8),

                  child: AspectRatio(

                    aspectRatio: 1,

                    child: Container(

                      decoration: BoxDecoration(

                        color: const Color(0xFFFFF8E1),

                        border: Border.all(color: scheme.outline, width: 2),

                        borderRadius: BorderRadius.circular(6),

                      ),

                      child: InteractiveViewer(

                        maxScale: 3,

                        child: _buildBoard(),

                      ),

                    ),

                  ),

                ),

              ),

            ),

          ],

        ),

      ),

    );

  }

 

  Widget _buildBoard() {

    final size = game.size;

    final winLine = game.winningLine;

    final last = game.lastMove;

    return GridView.builder(

      physics: const NeverScrollableScrollPhysics(),

      padding: EdgeInsets.zero,

      gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(

        crossAxisCount: size,

      ),

      itemCount: size * size,

      itemBuilder: (context, index) {

        final r = index ~/ size, c = index % size;

        final cell = game.board[r][c];

        final isWin = winLine.any((w) => w.x == r && w.y == c);

        final isLast = last != null && last.x == r && last.y == c;

        return GestureDetector(

          onTap: () => _onTap(r, c),

          child: Container(

            decoration: BoxDecoration(

              color: isWin

                  ? Colors.green.shade200

                  : isLast

                      ? Colors.amber.shade200

                      : null,

              border: Border.all(color: Colors.brown.shade200, width: 0.5),

            ),

            child: FittedBox(

              child: Padding(

                padding: const EdgeInsets.all(2),

                child: Text(

                  cell.symbol,

                  style: TextStyle(

                    fontWeight: FontWeight.bold,

                    color: cell == Player.x ? Colors.blue : Colors.red,

                  ),

                ),

              ),

            ),

          ),

        );

      },

    );

  }

}

 

class _ScoreChip extends StatelessWidget {

  final String label;

  final int score;

  final Color color;

  final bool active;

 

  const _ScoreChip({

    required this.label,

    required this.score,

    required this.color,

    required this.active,

  });

 

  @override

  Widget build(BuildContext context) {

    return AnimatedContainer(

      duration: const Duration(milliseconds: 200),

      padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),

      decoration: BoxDecoration(

        color: active ? color.withValues(alpha: 0.15) : Colors.transparent,

        border: Border.all(color: active ? color : Colors.grey.shade300, width: 2),

        borderRadius: BorderRadius.circular(20),

      ),

      child: Column(

        children: [

          Text(label, style: TextStyle(color: color, fontWeight: FontWeight.bold)),

          Text('$score', style: const TextStyle(fontSize: 18)),

        ],

      ),

    );

  }

}
